romulan_dewa_initializer = {
	name = "NAME_Dewa"
	# Dewa begins as a normal Romulan Star Empire outpost.  Keep the identity
	# and CP26's lookup key in one declaration: duplicate `flags` entries can
	# replace the first set during initializer parsing.
	flags = { outpost romulan_system romulan dewa }
	class = "rl_all_stars"
	usage = static_system_init

	planet = {
		class = star
		orbit_distance = 0
		orbit_angle = 0
		size = { min = 20 max = 30 }
		has_ring = no

		init_effect = {
			log = "Dewa initializer: Creating Dewa star"
			save_global_event_target_as = dewa_star
		}
	}
	
	change_orbit = 50
	
	planet = {
		count = { min = 2 max = 7 }
		class = rl_unhabitable_planets
		orbit_distance = { min = 20 max = 30 }
		home_planet = no
	}

	# New Romulus / Mol'Rihan is intentionally added to Infinite's own Dewa
	# initializer.  The Stellaris mod supplied the setting reference only.
	planet = {
		name = "NAME_CP26_Mol_Rihan"
		count = 1
		class = pc_continental
		subclass = psc_class_m
		orbit_distance = { min = 20 max = 30 }
		size = 16
		home_planet = no
		init_effect = {
			prevent_anomaly = yes
			set_planet_flag = cp26_mol_rihan
		}
		moon = {
			name = "NAME_CP26_Mol_Rihan_a"
			class = pc_barren_cold
			size = 6
			orbit_distance = 10
		}
		moon = {
			name = "NAME_CP26_Mol_Rihan_b"
			class = pc_barren_cold
			size = 3
			orbit_distance = 7
		}
	}

	late_init_effect = {
		create_starbase = {
			# Dewa is an established Romulan system before the supernova and the
			# future seat of the Republic, so it starts with a functional starport.
			size = starbase_starport
			owner = event_target:romulan_country
		}
		if = {
			limit = { exists = starbase }
			starbase = {
				set_starbase_module = {
					slot = 1
					module = shipyard
				}
			}
		}
	}
}
