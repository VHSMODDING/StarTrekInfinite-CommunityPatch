# Dedicated Bajor-Idran wormhole. The AI sees it as a normal route and can
# therefore use its ordinary exploration, military and civilian pathfinding.
# Human players keep the explicit instant fleet command.
cp26_bajoran_wormhole = {
	name = NATURAL_WORMHOLE_GALACTIC
	is_pathfind = yes
	uses_action = no
	requires_exploration = yes
	always_active = yes
	windup_time = 0
	winddown_time = 0
	ftl_multiplier = 0
	extends_sensors = yes
	icon_frame = 12
	remember_instances = yes
	animation_time = 1
	galactic_map_tooltip_header = NATURAL_WORMHOLE_GALACTIC
	galactic_map_tooltip_desc = NATURAL_WORMHOLE_DESC
	connection_type = one_to_one
	ftl_ship_effect = "wormhole_ftl_ship_effect_entity"
	ftl_bypass_effect = "wormhole_ftl_bypass_effect_entity"

	potential = { always = yes }

	country_can_use = {
		custom_tooltip = {
			fail_text = CP26_BAJORAN_WORMHOLE_USE_FLEET_COMMAND
			is_ai = yes
			OR = {
				is_cp26_dominion = yes
				has_country_flag = cp26_bajoran_wormhole_discovered
				has_seen_any_bypass = cp26_bajoran_wormhole
			}
			OR = {
				NOT = { exists = from }
				from = { NOT = { has_closed_borders = root } }
				from = { is_at_war_with = root }
				from = {
					NOT = { is_country_type = default }
					NOT = { is_country_type = cp26_dominion }
				}
			}
		}
	}

	# The route is native for AI planning, but the actual crossing is completed
	# by CP26's instant scripted transfer when the fleet enters the bypass.
	on_action = "on_entering_cp26_bajoran_wormhole"
}
