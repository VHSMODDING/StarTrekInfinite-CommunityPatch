# Dedicated Bajor-Idran natural wormhole. It remains separate from ordinary
# wormholes so its long-distance travel factor can be tuned independently.
cp26_bajoran_wormhole = {
	name = NATURAL_WORMHOLE_GALACTIC
	# Passage is handled by the explicit fleet command. Keeping this out of
	# automatic pathfinding prevents the engine from starting its slow generic
	# bypass travel animation instead of the instantaneous scripted transfer.
	is_pathfind = no
	uses_action = no
	requires_exploration = yes
	always_active = yes
	windup_time = 0
	winddown_time = 0
	ftl_multiplier = 0
	extends_sensors = yes
	icon_frame = 12
	remember_instances = yes
	animation_time = 1
	galactic_map_tooltip_header = NATURAL_WORMHOLE_GALACTIC
	galactic_map_tooltip_desc = NATURAL_WORMHOLE_DESC
	connection_type = one_to_one
	ftl_ship_effect = "wormhole_ftl_ship_effect_entity"
	ftl_bypass_effect = "wormhole_ftl_bypass_effect_entity"

	potential = { always = yes }

	country_can_use = {
		custom_tooltip = {
			fail_text = CP26_BAJORAN_WORMHOLE_USE_FLEET_COMMAND
			always = no
		}
	}

	on_action = "on_entering_wormhole"
}
