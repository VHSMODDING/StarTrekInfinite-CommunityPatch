# Fleet scope. Used by AI fleets after reaching either scripted endpoint.
cp26_ai_cross_bajoran_wormhole = {
	if = {
		limit = {
			exists = owner
			owner = { is_ai = yes }
			exists = event_target:bajoran_system
			exists = event_target:cp26_gamma_idran_system
			is_in_combat = no
			is_mia = no
			OR = {
				solar_system = { has_star_flag = cp26_gamma_idran_system }
				solar_system = { has_star_flag = bajoran_system }
			}
		}
		if = {
			limit = { solar_system = { has_star_flag = cp26_gamma_idran_system } }
			set_location = {
				target = event_target:bajoran_system
				distance = 25
				angle = random
				direction = in_system
			}
		}
		else = {
			set_location = {
				target = event_target:cp26_gamma_idran_system
				distance = 25
				angle = random
				direction = in_system
			}
		}
		clear_orders = yes
		clear_fleet_actions = this
	}
}
