namespace = cp26_romulan_split

# Start-gate for the vanilla supernova chain.  This is deliberately a separate
# event rather than an override of romulan_supernova.1: Infinite rejects event
# ID duplicates.  Once this gate has run, changing the setup option later
# cannot retroactively add or remove the arc for this campaign.
country_event = {
	id = cp26_romulan_split.0
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = romulan
		NOT = { has_country_flag = cp26_supernova_start_gate_resolved }
		NOT = { has_global_flag = cp26_block_supernova_arc }
		exists = event_target:romulan_homeworld
	}

	immediate = {
		set_country_flag = cp26_supernova_start_gate_resolved
		country_event = { id = romulan_supernova.1 }
		# Vanilla: 20 years to detection, then 24 to destruction.  This follows
		# the explosion by one day and only exists in enabled campaigns.
		country_event = { id = cp26_romulan_split.1 days = 15841 }
	}
}

# The vanilla supernova chain remains authoritative.  This follows it by one
# day, checks its romulan_homeworld_rip flag, then establishes the two canon
# successor states on planets that exist in Infinite's static map.
country_event = {
	id = cp26_romulan_split.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = romulan
		has_country_flag = romulan_homeworld_rip
		NOT = { has_country_flag = cp26_romulan_successors_created }
		exists = event_target:reman_species
	}

	immediate = {
		set_country_flag = cp26_romulan_successors_created
		# Infinite's vanilla Romulan supernova leaves an F-class placeholder.
		# CP26 changes only the destroyed Romulus system into a black-hole remnant;
		# it deliberately does not override the global vanilla supernova effect.
		random_galaxy_planet = {
			limit = { has_planet_flag = planet_romulus }
			solar_system = {
				set_star_class = sc_black_hole
				# Romulus is a binary system.  `star` is its central star;
				# converting every star would visibly create two black holes.  The
				# companion is removed so the remnant is one central black hole.
				star = {
					set_planet_flag = cp26_romulus_black_hole
					change_pc = pc_black_hole
				}
				every_system_planet = {
					limit = {
						is_star = yes
						NOT = { has_planet_flag = cp26_romulus_black_hole }
					}
					remove_planet = yes
				}
			}
		}
		# Keep an explicit handle to the pre-schism major country.  The choice
		# event later changes scopes several times; using this target prevents
		# Mol'Rihan from remaining with the temporary successor country.
		save_global_event_target_as = cp26_original_romulan_country
		owner_main_species = { save_event_target_as = cp26_romulan_species }

		random_galaxy_planet = {
			limit = { has_planet_flag = cp26_rator_iii }
			save_global_event_target_as = cp26_rator_iii
		}
		if = {
			limit = { exists = event_target:cp26_rator_iii }
			event_target:cp26_rator_iii.solar_system = {
				save_global_event_target_as = cp26_rator_system
			}
		}
		random_galaxy_planet = {
			limit = { has_planet_flag = cp26_mol_rihan }
			save_global_event_target_as = cp26_mol_rihan
		}

		# The original country persists as the Romulan Star Empire at Rator III.
		if = {
			limit = { exists = event_target:cp26_rator_iii }
			event_target:cp26_rator_iii = {
				set_owner = root
				if = { limit = { is_colony = no } create_colony = { owner = root } }
				set_capital = yes
				while = {
					count = 4
					create_pop = { species = event_target:reman_species }
				}
			}
		}
		# Re-establish the Star Empire's operational core at Rator III.  This
		# uses Infinite's own safe starter-fleet initializer: science ship with
		# scientist, constructor, and a two-corvette military patrol.
		give_technology = { tech = tech_corvette_unlock message = no }
		country_event = { id = game_start.9 }

		# The Republic is deliberately a native Infinite default country.  It does
		# not import Stellaris-only governments, scripted effects, or assumptions.
		if = {
			limit = { exists = event_target:cp26_mol_rihan }
			create_country = {
				name = "NAME_CP26_Romulan_Republic"
				adjective = "ADJ_CP26_Romulan_Republic"
				species = event_target:cp26_romulan_species
				country_id = "cp26_romulan_republic"
				type = default
				ship_prefix = "IRW"
				name_list = "RML1"
				# The Republic follows the source mod's post-imperial concept, but
				# uses only governments/civics that exist in Star Trek: Infinite.
				authority = auth_democratic
				civics = { civic = civic_higher_education }
				civics = { civic = civic_diplomatic_expertise }
				origin = origin_separatists
				flag = {
					# CP26's own white emblem mask.  The matching main/map/small/
					# bottombar DDS variants live under flags/emblem/.
					icon = { category = emblem file = "flag_cp26_romulan_republic.dds" }
					background = { category = backgrounds file = "00_solid.dds" }
					# A brighter post-imperial turquoise, distinct from the Star Empire's
					# muted native Romulan green in the galaxy map.
			colors = { "Romulan" "black" "null" "null" }
				}
				ethos = {
					ethic = ethic_militarist
					ethic = ethic_materialist
				}
				ignore_initial_colony_error = yes
			}
			last_created_country = {
				# Both graphical cultures are required.  The first governs ships, the
				# second governs cities and portrait presentation.
				set_graphical_culture = romulan_01
				set_city_graphical_culture = romulan_01
				set_country_flag = cp26_romulan_republic
				save_global_event_target_as = cp26_romulan_republic

				# The dynamic generic Romulan portrait loses its clothing layer in
				# Infinite.  unique_04 is the native, fully-rigged Romulan uniformed
				# portrait used by the vanilla Romulan leader events.
				create_leader = {
					class = ruler
					species = owner_main_species
					gender = male
					portrait = unique_04
					name = random
				}
				set_leader = last_created_leader
			}

			# A country created mid-campaign is removed by Infinite if it reaches
			# the next cleanup without a colony.  Give the provisional Republic
			# Mol'Rihan immediately, before the player-choice event can fire.
			# If the player selects the Republic, event .4 transfers this entire
			# local system back to the original major country one day later.
			event_target:cp26_mol_rihan = {
				set_owner = event_target:cp26_romulan_republic
				set_controller = event_target:cp26_romulan_republic
				if = { limit = { is_colony = no } create_colony = { owner = event_target:cp26_romulan_republic } }
				set_capital = yes
				solar_system = {
					every_system_planet = {
						limit = { is_colony = yes }
						set_owner = event_target:cp26_romulan_republic
						set_controller = event_target:cp26_romulan_republic
					}
					if = {
						limit = { exists = starbase }
						starbase = { set_owner = event_target:cp26_romulan_republic }
					}
					else = {
						create_starbase = { size = starbase_outpost owner = event_target:cp26_romulan_republic }
					}
				}
			}

			event_target:cp26_romulan_republic = {
				add_resource = { energy = 500 minerals = 300 food = 300 alloys = 200 }
				# It inherits the complete pre-schism research record, rather than a
				# starter package, so both successor states remain era-appropriate.
				copy_techs_from = { target = root }
				establish_communications_no_message = root
				# Mol'Rihan is assigned above solely to keep this country alive.  The
				# selected successor receives the final capital in its option.
			}
			establish_communications_no_message = event_target:cp26_romulan_republic

			# The Republic inherits the diplomatic intelligence network of the
			# pre-schism Empire: every state already known to the Star Empire is
			# contacted in both directions, without discovery notifications.
			every_country = {
				limit = {
					NOT = { is_same_value = root }
					has_communications = root
				}
				save_event_target_as = cp26_republic_contact
				establish_contact = { who = event_target:cp26_romulan_republic hidden = yes }
				establish_communications_no_message = event_target:cp26_romulan_republic
				event_target:cp26_romulan_republic = {
					establish_contact = { who = event_target:cp26_republic_contact hidden = yes }
					establish_communications_no_message = event_target:cp26_republic_contact
				}
			}

		# The Republic is the Federation-facing successor.  Both countries get
		# lasting initial trust and open borders, while the Star Empire remains
		# diplomatically separate and keeps its Tal Shiar behaviour.
		if = {
			limit = { exists = event_target:ufop_country }
			event_target:cp26_romulan_republic = {
				establish_communications_no_message = event_target:ufop_country
				set_closed_borders = { who = event_target:ufop_country status = no }
				add_trust = { amount = 75 who = event_target:ufop_country }
				add_opinion_modifier = {
					who = event_target:ufop_country
					modifier = opinion_cp26_romulan_republic_refuge
				}
			}
			event_target:ufop_country = {
				set_closed_borders = { who = event_target:cp26_romulan_republic status = no }
				add_trust = { amount = 75 who = event_target:cp26_romulan_republic }
				add_opinion_modifier = {
					who = event_target:cp26_romulan_republic
					modifier = opinion_cp26_romulan_republic_refuge
				}
			}
		}
		}

		if = {
			limit = {
				is_ai = no
				exists = event_target:cp26_romulan_republic
			}
			# The supernova remnant is established first.  The player receives the
			# succession decision on the following day, rather than at the same
			# instant as the explosion.
			country_event = { id = cp26_romulan_split.2 days = 1 }
		}
	}
}

# A human Romulan player selects the successor state; AI Romulans simply keep
# the Star Empire, while the Republic receives normal AI control.
country_event = {
	id = cp26_romulan_split.2
	title = cp26_romulan_split.2.name
	desc = cp26_romulan_split.2.desc
	picture = GFX_evt_shattered_planet_space_valdore_romulan
	show_sound = event_terrifying_supernova_explosion
	is_triggered_only = yes

	option = {
		name = cp26_romulan_split.2.a
		hidden_effect = {
			# The player keeps the original Star Empire.  Confirm the provisional
			# Republic's Mol'Rihan capital and its system station.
			event_target:cp26_mol_rihan = {
				set_owner = event_target:cp26_romulan_republic
				set_controller = event_target:cp26_romulan_republic
				if = { limit = { is_colony = no } create_colony = { owner = event_target:cp26_romulan_republic } }
				set_capital = yes
				while = {
					count = 4
					create_pop = { species = event_target:reman_species }
				}
				solar_system = {
					if = {
						limit = { exists = starbase }
						starbase = { set_owner = event_target:cp26_romulan_republic }
					}
					else = {
						create_starbase = { size = starbase_outpost owner = event_target:cp26_romulan_republic }
					}
				}
			}
			event_target:cp26_romulan_republic = {
				give_technology = { tech = tech_corvette_unlock message = no }
				country_event = { id = game_start.9 }
			}
			country_event = { id = cp26_romulan_split.3 days = 1 }
		}
	}
	option = {
		name = cp26_romulan_split.2.b
		trigger = { exists = event_target:cp26_romulan_republic }
		hidden_effect = {
			# The major-country flag is fixed while generating the galaxy.  The
			# existing major country therefore becomes the selected Republic: it
			# keeps the major-map rendering, researched technology and vanilla
			# Romulan quest tree/progress.
			set_name = "NAME_CP26_Romulan_Republic"
			set_adjective = "ADJ_CP26_Romulan_Republic"
			set_country_flag = cp26_romulan_republic_major
			# Unlike script flags, this replaces the actual visual flag of the
			# original major country.  The player therefore sees the Republic
			# emblem after choosing the Republic, including on the galaxy map.
			change_country_flag = {
				icon = { category = emblem file = "flag_cp26_romulan_republic.dds" }
				background = { category = backgrounds file = "00_solid.dds" }
				colors = { "Romulan" "black" "null" "null" }
			}
			# The Republic keeps the major country's research, tree and visual
			# identity, but not its State Senate / Tal Shiar civic package.
			if = { limit = { has_civic = civic_closet_politics } force_remove_civic = civic_closet_politics }
			if = { limit = { has_civic = civic_state_vigilance } force_remove_civic = civic_state_vigilance }
			force_add_civic = civic_higher_education
			force_add_civic = civic_diplomatic_expertise
			# The original Empire begins authoritarian.  The elected Republic needs
			# a genuine post-imperial political basis before changing authority.
			if = { limit = { has_ethic = ethic_authoritarian } country_remove_ethic = ethic_authoritarian }
			country_add_ethic = ethic_egalitarian
			change_government = { authority = auth_cp26_romulan_republic }

			event_target:cp26_romulan_republic = {
				# The runtime-created state becomes the AI Star Empire at Rator III.
				set_name = "NAME_CP26_Romulan_Star_Empire"
				set_adjective = "ADJ_CP26_Romulan_Star_Empire"
				set_country_flag = cp26_romulan_star_empire_successor
				# This country was initially generated as the Republic.  Reset its
				# visible flag when it becomes the AI Star Empire.
				change_country_flag = {
					icon = { category = emblem file = "flag_Romulan_12.dds" }
					background = { category = backgrounds file = "00_solid.dds" }
					colors = { "Romulan" "black" "null" "null" }
				}
				if = { limit = { has_civic = civic_higher_education } force_remove_civic = civic_higher_education }
				if = { limit = { has_civic = civic_diplomatic_expertise } force_remove_civic = civic_diplomatic_expertise }
				force_add_civic = civic_closet_politics
				force_add_civic = civic_state_vigilance
				change_government = { authority = auth_cp26_romulan_star_empire }
			}

			if = {
				limit = { exists = event_target:cp26_rator_iii }
				event_target:cp26_rator_iii = {
					set_owner = event_target:cp26_romulan_republic
					set_controller = event_target:cp26_romulan_republic
					if = { limit = { is_colony = no } create_colony = { owner = event_target:cp26_romulan_republic } }
					set_capital = yes
					solar_system = {
						if = {
							limit = { exists = starbase }
							starbase = { set_owner = event_target:cp26_romulan_republic }
						}
					}
				}
			}
			event_target:cp26_romulan_republic = {
				give_technology = { tech = tech_corvette_unlock message = no }
				country_event = { id = game_start.9 }
			}
			if = {
				limit = { exists = event_target:cp26_rator_system }
				event_target:cp26_romulan_republic = {
					every_owned_fleet = {
						set_location = event_target:cp26_rator_system
					}
				}
			}
			if = {
				limit = { exists = event_target:cp26_mol_rihan }
				event_target:cp26_mol_rihan = {
					# This option is evaluated by the original major country, which
					# has just become the Republic.  Do not use the global target
					# here: create_country can overwrite its runtime scope with the
					# temporary Star Empire (and wrongly gives it Dewa).
					set_owner = root
					set_controller = root
					if = { limit = { is_colony = no } create_colony = { owner = root } }
					set_capital = yes
					while = {
						count = 4
						create_pop = { species = event_target:reman_species }
					}
					solar_system = {
						if = {
							limit = { exists = starbase }
							starbase = { set_owner = root }
						}
					}
				}
			}
			# Ownership and capitals are recalculated asynchronously by Infinite.
			# Finalise the transfer on the selected major country itself one day
			# later, so the former Rator capital cannot be retained.
			country_event = { id = cp26_romulan_split.4 days = 1 }
			country_event = { id = cp26_romulan_split.3 days = 1 }
		}
	}
}

# Final capital hand-off for a player who selected the Republic.
country_event = {
	id = cp26_romulan_split.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = cp26_romulan_republic_major
	}

	immediate = {
		# Do a fresh lookup instead of using an event target saved before the
		# country switch.  That target can resolve to the temporary AI country
		# after a one-day delay, leaving Dewa as a Star Empire enclave.
		random_galaxy_planet = {
			limit = { has_planet_flag = cp26_mol_rihan }
			set_owner = root
			set_controller = root
			set_capital = yes
			solar_system = {
				# Dewa is Mol'Rihan's system.  Transfer its whole local
				# administration to the Republic, not merely its capital planet.
				every_system_planet = {
					limit = { is_colony = yes }
					set_owner = root
					set_controller = root
				}
				if = {
					limit = { exists = starbase }
					starbase = { set_owner = root }
				}
				else = {
					create_starbase = {
						size = starbase_outpost
						owner = root
					}
				}
			}
		}
	}
}

# One-shot repair command for a save in which the vanilla supernova has
# already occurred.  It is intentionally independent from the succession
# event so it cannot create either successor country a second time.

	id = cp26_romulan_split.5
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		random_galaxy_planet = {
			limit = { has_planet_flag = planet_romulus }
			solar_system = {
				set_star_class = sc_black_hole
				star = {
					set_planet_flag = cp26_romulus_black_hole
					change_pc = pc_black_hole
				}
				every_system_planet = {
					limit = {
						is_star = yes
						NOT = { has_planet_flag = cp26_romulus_black_hole }
					}
					remove_planet = yes
				}
			}
		}
	}
}
