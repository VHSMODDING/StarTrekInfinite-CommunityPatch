namespace = cp26_gamma

# Rebuild the Gamma pocket after procedural hyperlane generation. This removes
# any accidental bridge to Alpha/Beta and then creates a deterministic internal
# warp network between the six Gamma systems.
event = {
	id = cp26_gamma.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		exists = event_target:cp26_gamma_idran_system
		exists = event_target:cp26_gamma_new_bajor_system
		exists = event_target:cp26_gamma_karemma_system
		exists = event_target:cp26_gamma_teplan_system
		exists = event_target:cp26_gamma_rondac_system
		exists = event_target:cp26_gamma_omarion_system
	}

	immediate = {
		# Remove every generated connection, including any procedural bridge.
		event_target:cp26_gamma_idran_system = { isolate_system = yes }
		event_target:cp26_gamma_new_bajor_system = { isolate_system = yes }
		event_target:cp26_gamma_karemma_system = { isolate_system = yes }
		event_target:cp26_gamma_teplan_system = { isolate_system = yes }
		event_target:cp26_gamma_rondac_system = { isolate_system = yes }
		event_target:cp26_gamma_omarion_system = { isolate_system = yes }

		# Deterministic internal warp routes. No route leaves this graph.
		event_target:cp26_gamma_idran_system = {
			add_hyperlane = { from = this to = event_target:cp26_gamma_new_bajor_system }
			add_hyperlane = { from = this to = event_target:cp26_gamma_karemma_system }
		}
		event_target:cp26_gamma_new_bajor_system = {
			add_hyperlane = { from = this to = event_target:cp26_gamma_teplan_system }
		}
		event_target:cp26_gamma_karemma_system = {
			add_hyperlane = { from = this to = event_target:cp26_gamma_rondac_system }
		}
		event_target:cp26_gamma_teplan_system = {
			add_hyperlane = { from = this to = event_target:cp26_gamma_rondac_system }
		}
		event_target:cp26_gamma_rondac_system = {
			add_hyperlane = { from = this to = event_target:cp26_gamma_omarion_system }
		}
		set_global_flag = cp26_gamma_quadrant_initialized
	}
}
