##########################################################################
# 0 - Dispatch : choisit l'event faction approprié
##########################################################################
country_event = {
	id = cp26_bajoran_wormhole.0
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_bajoran_wormhole_dispatched }
		OR = {
			is_ufop = yes
			is_klingon = yes
			is_cardassian = yes
			is_romulan = yes
			is_ferengi_empire = yes
		}
	}

	immediate = {
		set_country_flag = cp26_bajoran_wormhole_dispatched
		if = {
			limit = { is_ufop = yes }
			country_event = { id = cp26_bajoran_wormhole.10 }
		}
		else_if = {
			limit = { is_klingon = yes }
			country_event = { id = cp26_bajoran_wormhole.20 }
		}
		else_if = {
			limit = { is_cardassian = yes }
			country_event = { id = cp26_bajoran_wormhole.30 }
		}
		else_if = {
			limit = { is_romulan = yes }
			country_event = { id = cp26_bajoran_wormhole.40 }
		}
		else_if = {
			limit = { is_ferengi_empire = yes }
			country_event = { id = cp26_bajoran_wormhole.50 }
		}
	}
}

##########################################################################
# 110 - Commande de flotte : traversée instantanée Bajor <-> Idran
##########################################################################
fleet_event = {
	id = cp26_bajoran_wormhole.110
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = owner
		owner = { has_country_flag = cp26_bajoran_wormhole_discovered }
		has_global_flag = cp26_bajoran_wormhole_spawned
		exists = event_target:bajoran_system
		exists = event_target:cp26_gamma_idran_system
		is_in_combat = no
		is_mia = no
		OR = {
			solar_system = { has_star_flag = bajoran_system }
			solar_system = { has_star_flag = cp26_gamma_idran_system }
		}
	}

	immediate = {
		if = {
			limit = { solar_system = { has_star_flag = bajoran_system } }
			set_location = {
				target = event_target:cp26_gamma_idran_system
				distance = 25
				angle = random
				direction = in_system
			}
		}
		else = {
			set_location = {
				target = event_target:bajoran_system
				distance = 25
				angle = random
				direction = in_system
			}
		}
		clear_orders = yes
		clear_fleet_actions = this
	}
}

##########################################################################
# 120 - Dominion IA : envoie une flotte de guerre Gamma vers Idran
##########################################################################
country_event = {
	id = cp26_bajoran_wormhole.120
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		is_cp26_dominion = yes
		is_at_war = yes
		has_global_flag = cp26_bajoran_wormhole_spawned
		exists = event_target:bajoran_system
		exists = event_target:cp26_gamma_idran_system
		any_owned_fleet = {
			is_ship_class = shipclass_military
			is_in_combat = no
			is_mia = no
			solar_system = { has_star_flag = cp26_gamma_quadrant_system }
		}
	}

	immediate = {
		# Resolve the star every time so this also works on saves created before
		# the scripted transit command was added.
		event_target:cp26_gamma_idran_system = {
			random_system_planet = {
				limit = { is_star = yes }
				save_global_event_target_as = cp26_gamma_idran_transit_star
			}
		}

		if = {
			limit = { exists = event_target:cp26_gamma_idran_transit_star }
			random_owned_fleet = {
				limit = {
					is_ship_class = shipclass_military
					is_in_combat = no
					is_mia = no
					solar_system = { has_star_flag = cp26_gamma_quadrant_system }
				}
				if = {
					limit = { solar_system = { has_star_flag = cp26_gamma_idran_system } }
					cp26_ai_dominion_cross_bajoran_wormhole = yes
				}
				else = {
					auto_move_to_planet = {
						target = event_target:cp26_gamma_idran_transit_star
						clear_auto_move_on_arrival = yes
						arrival_effect = cp26_ai_dominion_cross_bajoran_wormhole
					}
				}
			}
		}
	}
}

##########################################################################
# 100 - Découverte du trou de ver (tous empires)
##########################################################################
country_event = {
	id = cp26_bajoran_wormhole.100
	title = cp26_bajoran_wormhole.100.name
	desc = cp26_bajoran_wormhole.100.desc
	picture = GFX_evt_highway_ftl_travel_ufop
	show_sound = event_alien_nature
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = cp26_bajoran_wormhole_discovered }
	}

	immediate = {
		set_country_flag = cp26_bajoran_wormhole_discovered
		end_event_chain = cp26_bajoran_wormhole_chain
		add_seen_bypass_type = cp26_bajoran_wormhole
		set_country_flag = encountered_first_wormhole

		# Create and explicitly link both natural-wormhole endpoints.
		if = {
			limit = {
				NOT = { has_global_flag = cp26_bajoran_wormhole_spawned }
				exists = event_target:bajoran_system
				exists = event_target:cp26_gamma_idran_system
			}
			event_target:bajoran_system = {
				if = {
					limit = { has_natural_wormhole = no }
					spawn_natural_wormhole = {
						bypass_type = cp26_bajoran_wormhole
						random_pos = no
						orbit_angle = 100
						orbit_distance = -150
					}
				}
			}
			event_target:cp26_gamma_idran_system = {
				if = {
					limit = { has_natural_wormhole = no }
					spawn_natural_wormhole = {
						bypass_type = cp26_bajoran_wormhole
						random_pos = no
						orbit_angle = 360
						orbit_distance = 20
					}
				}
				link_wormholes = event_target:bajoran_system
			}
			set_global_flag = cp26_bajoran_wormhole_spawned
		}

		# Spawn the Dominion beyond the newly opened passage without granting
		# communications. It remains unknown until normal first contact begins.
		if = {
			limit = {
				NOT = { has_global_flag = cp26_dominion_spawned }
				exists = event_target:cp26_gamma_omarion_system
			}
			country_event = { id = cp26_dominion.1 }
		}
	}

	# ── UFOP ──────────────────────────────────────────────────────────────
	option = {
		name = cp26_bajoran_wormhole.100.a_ufop
		trigger = { is_ufop = yes }
	}
	option = {
		name = cp26_bajoran_wormhole.100.b_ufop
		trigger = { is_ufop = yes }
	}

	# ── Klingon ───────────────────────────────────────────────────────────
	option = {
		name = cp26_bajoran_wormhole.100.a_klingon
		trigger = { is_klingon = yes }
	}
	option = {
		name = cp26_bajoran_wormhole.100.b_klingon
		trigger = { is_klingon = yes }
	}

	# ── Cardassien ────────────────────────────────────────────────────────
	option = {
		name = cp26_bajoran_wormhole.100.a_cardassian
		trigger = { is_cardassian = yes }
	}
	option = {
		name = cp26_bajoran_wormhole.100.b_cardassian
		trigger = { is_cardassian = yes }
	}

	# ── Romulien ──────────────────────────────────────────────────────────
	option = {
		name = cp26_bajoran_wormhole.100.a_romulan
		trigger = { is_romulan = yes }
	}
	option = {
		name = cp26_bajoran_wormhole.100.b_romulan
		trigger = { is_romulan = yes }
	}

	# ── Ferengi ───────────────────────────────────────────────────────────
	option = {
		name = cp26_bajoran_wormhole.100.a_ferengi
		trigger = { is_ferengi_empire = yes }
	}
	option = {
		name = cp26_bajoran_wormhole.100.b_ferengi
		trigger = { is_ferengi_empire = yes }
	}
}
