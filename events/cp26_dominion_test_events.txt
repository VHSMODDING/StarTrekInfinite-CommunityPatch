##########################################################################
# 1 - Real campaign spawn, called when the Bajoran wormhole appears.
##########################################################################
country_event = {
	id = cp26_dominion.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { has_global_flag = cp26_dominion_spawned }
		exists = event_target:cp26_gamma_omarion_system
	}

	immediate = {
		set_global_flag = cp26_dominion_spawned

		# Save compatibility: older Gamma pockets were generated before the
		# Founders' homeworld was part of the Omarion initializer.
		if = {
			limit = { NOT = { exists = event_target:cp26_dominion_homeworld } }
			event_target:cp26_gamma_omarion_system = {
				random_system_planet = {
					limit = { is_star = no is_colony = no }
					change_pc = pc_ocean
					set_name = "NAME_CP26_Founders_Homeworld"
					set_planet_flag = cp26_dominion_homeworld
					prevent_anomaly = yes
					save_global_event_target_as = cp26_dominion_homeworld
				}
			}
		}

		event_target:cp26_gamma_omarion_system = {
			create_species = {
				class = "CP26_FOUNDER"
				portrait = "cp26_founder_female_01"
				name = "NAME_CP26_Founder_Species"
				plural = "NAME_CP26_Founder_Species_Plural"
				adjective = "ADJ_CP26_Founder_Species"
				name_list = "CP26_DOMINION"
				traits = {
					trait = "trait_cp26_founder_shapeshifter"
					trait = "trait_pc_ocean_preference"
				}
			}
			last_created_species = { save_global_event_target_as = cp26_founder_species }

			create_species = {
				class = "CP26_VORTA"
				portrait = "cp26_vorta_male_01"
				name = "NAME_CP26_Vorta_Species"
				plural = "NAME_CP26_Vorta_Species_Plural"
				adjective = "ADJ_CP26_Vorta_Species"
				name_list = "CP26_DOMINION"
				traits = {
					trait = "trait_cp26_vorta_cloned_administrator"
					trait = "trait_pc_ocean_preference"
				}
			}
			last_created_species = { save_global_event_target_as = cp26_vorta_species }

			create_species = {
				class = "CP26_JEMHADAR"
				portrait = "cp26_jemhadar_male_01"
				name = "NAME_CP26_JemHadar_Species"
				plural = "NAME_CP26_JemHadar_Species_Plural"
				adjective = "ADJ_CP26_JemHadar_Species"
				name_list = "CP26_DOMINION"
				traits = {
					trait = "trait_cp26_jemhadar_engineered_soldier"
					trait = "trait_pc_ocean_preference"
				}
			}
			last_created_species = { save_global_event_target_as = cp26_jemhadar_species }

			create_country = {
				name = "NAME_CP26_The_Dominion"
				species = event_target:cp26_founder_species
				country_id = "dominion"
				type = cp26_dominion
				ship_prefix = "DVS"
				name_list = "CP26_DOMINION"
				authority = auth_cp26_great_link
				civics = { civic = civic_cp26_great_link }
				civics = { civic = civic_cp26_engineered_castes }
				origin = "origin_default"
				flag = {
					icon = { category = "emblem" file = "flag_cp26_dominion.dds" }
					background = { category = "backgrounds" file = "00_solid.dds" }
					colors = { "dark_purple" "black" "null" "null" }
				}
				ethos = {
					ethic = ethic_authoritarian
					ethic = ethic_militarist
					ethic = ethic_xenophobe
				}
				ignore_initial_colony_error = yes
			}

			last_created_country = {
				set_adjective = "ADJ_CP26_Dominion"
				set_graphical_culture = dominion
				set_country_flag = cp26_dominion
				save_global_event_target_as = cp26_dominion_country
				create_leader = {
					class = ruler
					species = event_target:cp26_founder_species
					name = NAME_CP26_Female_Founder
					portrait = cp26_founder_female_01
					gender = female
					skill = 7
				}
				set_leader = last_created_leader
			}
		}

		if = {
			limit = { exists = event_target:cp26_dominion_country }
			event_target:cp26_dominion_country = {
				# The Dominion knows its own Idran endpoint. The AI-only native
				# route can consequently be planned like any other wormhole.
				set_country_flag = cp26_bajoran_wormhole_discovered
				set_country_flag = cp26_bajoran_wormhole_route_registered
				add_seen_bypass_type = cp26_bajoran_wormhole
				give_technology = { tech = tech_shipyard message = no }
				give_technology = { tech = tech_starbase_starport message = no }
				give_technology = { tech = tech_starbase_starhold message = no }
				give_technology = { tech = tech_planetary_government message = no }
				give_technology = { tech = tech_colonial_centralization message = no }
				give_technology = { tech = tech_corvette_unlock message = no }
				give_technology = { tech = tech_destroyer_unlock message = no }
				give_technology = { tech = tech_cruiser_unlock message = no }
				give_technology = { tech = tech_battleship_unlock message = no }
				give_technology = { tech = tech_phaser_3 message = no }
				give_technology = { tech = tech_torpedo_2 message = no }
				# Era-coherent support package.  The Dominion appears as an old
				# Gamma Quadrant power, not with the starter sensors, shields and
				# warp drive used by a 2345 empire.
				give_technology = { tech = tech_warp_drive_2 message = no }
				give_technology = { tech = tech_warp_drive_3 message = no }
				give_technology = { tech = tech_sensor_2 message = no }
				give_technology = { tech = tech_sensor_3 message = no }
				give_technology = { tech = tech_shield_2 message = no }
				give_technology = { tech = tech_shield_3 message = no }
				give_technology = { tech = tech_hull_2 message = no }
				give_technology = { tech = tech_hull_3 message = no }
				give_technology = { tech = tech_engine_2 message = no }
				give_technology = { tech = tech_engine_3 message = no }
				give_technology = { tech = tech_computer_2 message = no }
				give_technology = { tech = tech_computer_3 message = no }
				give_technology = { tech = tech_shield_booster message = no }
				give_technology = { tech = tech_warp_speed_increase_1 message = no }
				give_technology = { tech = tech_warp_speed_increase_2 message = no }
				give_technology = { tech = tech_cp26_dominion_attack_ship message = no }
				give_technology = { tech = tech_cp26_dominion_battle_cruiser message = no }
				give_technology = { tech = tech_cp26_dominion_polaron_beam message = no }
				give_technology = { tech = tech_cp26_dominion_polaron_torpedo message = no }
				give_technology = { tech = tech_cp26_jemhadar_cloning message = no }
				give_technology = { tech = tech_cp26_ketracel_white message = no }
				set_country_flag = cp26_dominion_era_tech_aligned

				create_ship_design = { design = "NAME_CP26_DominionAttackShip" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_CP26_DominionBattleCruiser" }
				add_ship_design = last_created_design
				# A campaign that reaches the wormhole late must not spawn a
				# 2370-era Dominion decades behind the rest of the galaxy.
				if = {
					limit = { years_passed > 28 }
					give_technology = { tech = tech_cp26_dominion_battleship message = no }
					create_ship_design = { design = "NAME_CP26_DominionBattleship" }
					add_ship_design = last_created_design
				}
				add_resource = {
					energy = 8000
					minerals = 8000
					alloys = 12000
					food = 6000
					influence = 500
					unity = 3000
				}
			}

			event_target:cp26_dominion_homeworld = {
				create_colony = { owner = event_target:cp26_dominion_country }

				set_colony_type = col_default
				set_capital = yes
				clear_blockers = yes

				while = { count = 5 create_pop = { species = event_target:cp26_vorta_species } }
				while = { count = 8 create_pop = { species = event_target:cp26_jemhadar_species } }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_industrial }
				add_district_and_planet_size_if_needed_effect = { district = district_industrial }
				add_district_and_planet_size_if_needed_effect = { district = district_generator }
				add_district_and_planet_size_if_needed_effect = { district = district_mining }
				add_district_and_planet_size_if_needed_effect = { district = district_farming }
				add_building = building_capital
				add_building = building_foundry_1
				add_building = building_research_lab_1
				add_building = building_stronghold
			}

			event_target:cp26_gamma_omarion_system = {
				create_starbase = {
					size = starbase_starport
					owner = event_target:cp26_dominion_country
				}
				starbase = { set_starbase_module = { slot = 1 module = shipyard } }
			}

			event_target:cp26_dominion_country = {
				create_fleet = {
					name = "NAME_CP26_Dominion_First_Fleet"
					effect = {
						while = { count = 6 create_ship = { name = random design = "NAME_CP26_DominionAttackShip" graphical_culture = "dominion_01" } }
						while = { count = 2 create_ship = { name = random design = "NAME_CP26_DominionBattleCruiser" graphical_culture = "dominion_01" } }
						if = { limit = { years_passed > 28 } create_ship = { name = random design = "NAME_CP26_DominionBattleship" graphical_culture = "dominion_01" } }
						set_owner = prev
						set_location = event_target:cp26_gamma_omarion_system
					}
				}
				create_fleet = {
					name = "NAME_CP26_Dominion_Second_Fleet"
					effect = {
						while = { count = 5 create_ship = { name = random design = "NAME_CP26_DominionAttackShip" graphical_culture = "dominion_01" } }
						while = { count = 2 create_ship = { name = random design = "NAME_CP26_DominionBattleCruiser" graphical_culture = "dominion_01" } }
						if = { limit = { years_passed > 28 } create_ship = { name = random design = "NAME_CP26_DominionBattleship" graphical_culture = "dominion_01" } }
						set_owner = prev
						set_location = event_target:cp26_gamma_omarion_system
					}
				}
			}
		}
	}

##########################################################################
# 5 - Bootstrap civil du Dominion.
#
# Les pays crees dynamiquement recoivent bien les modeles civils generiques,
# mais l'IA d'Infinite ne lance pas toujours leur premiere construction. Ce
# filet de securite ne s'execute qu'une fois et complete uniquement les
# effectifs manquants. La gestion normale de l'IA reprend ensuite.
##########################################################################
country_event = {
	id = cp26_dominion.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_cp26_dominion = yes
		NOT = { has_country_flag = cp26_dominion_civilian_bootstrap_complete }
		exists = capital_scope
	}

	immediate = {
		# Premier vaisseau scientifique et son officier.
		if = {
			limit = {
				count_owned_ship = {
					count < 1
					limit = { is_ship_class = shipclass_science_ship }
				}
			}
			create_fleet = {
				effect = {
					set_owner = prev
					create_ship = {
						name = random
						random_existing_design = science
					}
					set_fleet_stance = evasive
					set_location = prev.capital_scope.star
					owner = {
						create_leader = {
							class = scientist
							sub_type = survey
							name = random
							species = owner_main_species
						}
					}
					set_leader = last_created_leader
				}
			}
		}

		# Deuxieme scientifique : la poche Gamma comporte deux branches.
		if = {
			limit = {
				count_owned_ship = {
					count < 2
					limit = { is_ship_class = shipclass_science_ship }
				}
			}
			create_fleet = {
				effect = {
					set_owner = prev
					create_ship = {
						name = random
						random_existing_design = science
					}
					set_fleet_stance = evasive
					set_location = prev.capital_scope.star
					owner = {
						create_leader = {
							class = scientist
							sub_type = survey
							name = random
							species = owner_main_species
						}
					}
					set_leader = last_created_leader
				}
			}
		}

		# Deux constructeurs permettent de revendiquer les deux branches sans
		# remplacer les choix d'expansion de l'IA par un script.
		if = {
			limit = {
				count_owned_ship = {
					count < 1
					limit = { is_ship_class = shipclass_constructor }
				}
			}
			create_fleet = {
				effect = {
					set_owner = prev
					create_ship = {
						name = random
						random_existing_design = constructor
					}
					set_fleet_stance = evasive
					set_location = prev.capital_scope.star
				}
			}
		}
		if = {
			limit = {
				count_owned_ship = {
					count < 2
					limit = { is_ship_class = shipclass_constructor }
				}
			}
			create_fleet = {
				effect = {
					set_owner = prev
					create_ship = {
						name = random
						random_existing_design = constructor
					}
					set_fleet_stance = evasive
					set_location = prev.capital_scope.star
				}
			}
		}

		set_country_flag = cp26_dominion_civilian_bootstrap_complete
	}
}

##########################################################################
# 10 - Safe AI progression for the Dominion mission tree.
#
# Infinite 1.0.7's native quest-tree module is unsafe for a custom country
# created after game start (CQuestStateManager::MoveNode crash).  This event
# evaluates the same sixteen milestones and grants the same rewards without
# calling ForceNodeCompleted.
##########################################################################
country_event = {
	id = cp26_dominion.10
	hide_window = yes
	is_triggered_only = yes

	trigger = { is_cp26_dominion = yes }

	immediate = {
		# Root - The Great Link
		if = {
			limit = {
				NOT = { has_country_flag = cp26_dominion_mission_1 }
				has_technology = tech_cp26_dominion_polaron_beam
				has_technology = tech_cp26_jemhadar_cloning
			}
			set_country_flag = cp26_dominion_mission_1
			add_resource = { unity = 1500 influence = 100 }
			add_modifier = { modifier = qt_cp26_dominion_1_effect }
		}

		# Administration branch
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_1
				NOT = { has_country_flag = cp26_dominion_mission_2 }
				has_monthly_income = { resource = energy value >= 45 }
				has_monthly_income = { resource = minerals value >= 45 }
			}
			set_country_flag = cp26_dominion_mission_2
			add_resource = { energy = 2000 minerals = 2000 }
			add_modifier = { modifier = qt_cp26_dominion_2_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_2
				NOT = { has_country_flag = cp26_dominion_mission_3 }
				count_owned_planet = { count >= 3 }
			}
			set_country_flag = cp26_dominion_mission_3
			add_resource = { alloys = 2500 influence = 150 }
			add_modifier = { modifier = qt_cp26_dominion_3_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_3
				NOT = { has_country_flag = cp26_dominion_mission_4 }
				count_owned_planet = { count >= 5 }
				resource_stockpile_compare = { resource = influence value >= 300 }
			}
			set_country_flag = cp26_dominion_mission_4
			add_resource = { unity = 3000 influence = 200 }
			add_modifier = { modifier = qt_cp26_dominion_4_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_4
				NOT = { has_country_flag = cp26_dominion_mission_5 }
				count_owned_planet = { count >= 7 }
				has_monthly_income = { resource = unity value >= 50 }
			}
			set_country_flag = cp26_dominion_mission_5
			add_resource = { energy = 4000 minerals = 4000 unity = 4000 }
			add_modifier = { modifier = qt_cp26_dominion_5_effect }
		}

		# Jem'Hadar branch
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_1
				NOT = { has_country_flag = cp26_dominion_mission_6 }
				fleet_power >= 4000
			}
			set_country_flag = cp26_dominion_mission_6
			add_resource = { alloys = 2000 food = 2000 }
			add_modifier = { modifier = qt_cp26_dominion_6_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_6
				NOT = { has_country_flag = cp26_dominion_mission_7 }
				has_technology = tech_cp26_ketracel_white
				resource_stockpile_compare = { resource = food value >= 5000 }
			}
			set_country_flag = cp26_dominion_mission_7
			add_resource = { alloys = 2500 food = 3500 }
			add_modifier = { modifier = qt_cp26_dominion_7_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_7
				NOT = { has_country_flag = cp26_dominion_mission_8 }
				fleet_power >= 9000
				resource_stockpile_compare = { resource = alloys value >= 5000 }
			}
			set_country_flag = cp26_dominion_mission_8
			add_resource = { alloys = 4000 influence = 150 }
			add_modifier = { modifier = qt_cp26_dominion_8_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_8
				NOT = { has_country_flag = cp26_dominion_mission_9 }
				fleet_power >= 14000
				has_monthly_income = { resource = alloys value >= 75 }
			}
			set_country_flag = cp26_dominion_mission_9
			add_resource = { alloys = 5000 unity = 3000 }
			add_modifier = { modifier = qt_cp26_dominion_9_effect }
		}

		# Founder/Vorta intelligence branch
		set_variable = { which = cp26_dominion_known_empires value = 0 }
		every_country = {
			limit = {
				has_communications = root
				is_regular_empire = yes
				NOT = { is_same_value = root }
			}
			root = { change_variable = { which = cp26_dominion_known_empires value = 1 } }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_1
				NOT = { has_country_flag = cp26_dominion_mission_10 }
				check_variable = { which = cp26_dominion_known_empires value >= 2 }
			}
			set_country_flag = cp26_dominion_mission_10
			add_resource = { influence = 150 unity = 1500 }
			add_modifier = { modifier = qt_cp26_dominion_10_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_10
				NOT = { has_country_flag = cp26_dominion_mission_11 }
				successful_spy_mission_count >= 1
			}
			set_country_flag = cp26_dominion_mission_11
			add_resource = { influence = 200 unity = 2000 }
			add_modifier = { modifier = qt_cp26_dominion_11_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_11
				NOT = { has_country_flag = cp26_dominion_mission_12 }
				successful_spy_mission_count >= 3
				check_variable = { which = cp26_dominion_known_empires value >= 4 }
			}
			set_country_flag = cp26_dominion_mission_12
			add_resource = { influence = 250 unity = 2500 }
			add_modifier = { modifier = qt_cp26_dominion_12_effect }
		}
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_12
				NOT = { has_country_flag = cp26_dominion_mission_13 }
				successful_spy_mission_count >= 5
				resource_stockpile_compare = { resource = influence value >= 500 }
			}
			set_country_flag = cp26_dominion_mission_13
			add_resource = { influence = 300 unity = 3500 }
			add_modifier = { modifier = qt_cp26_dominion_13_effect }
		}

		# Convergence and heavy warship era
		if = {
			limit = {
				has_country_flag = cp26_dominion_mission_5
				has_country_flag = cp26_dominion_mission_9
				has_country_flag = cp26_dominion_mission_13
