namespace = cp26_dominion

##########################################################################
# 1 - Real campaign spawn, called when the Bajoran wormhole appears.
##########################################################################
country_event = {
	id = cp26_dominion.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		NOT = { has_global_flag = cp26_dominion_spawned }
		exists = event_target:cp26_gamma_omarion_system
	}

	immediate = {
		set_global_flag = cp26_dominion_spawned

		# Save compatibility: older Gamma pockets were generated before the
		# Founders' homeworld was part of the Omarion initializer.
		if = {
			limit = { NOT = { exists = event_target:cp26_dominion_homeworld } }
			event_target:cp26_gamma_omarion_system = {
				random_system_planet = {
					limit = { is_star = no is_colony = no }
					change_pc = pc_ocean
					set_name = "NAME_CP26_Founders_Homeworld"
					set_planet_flag = cp26_dominion_homeworld
					prevent_anomaly = yes
					save_global_event_target_as = cp26_dominion_homeworld
				}
			}
		}

		event_target:cp26_gamma_omarion_system = {
			create_species = {
				class = "CP26_FOUNDER"
				portrait = "cp26_founder_female_01"
				name = "NAME_CP26_Founder_Species"
				plural = "NAME_CP26_Founder_Species_Plural"
				adjective = "ADJ_CP26_Founder_Species"
				name_list = "CP26_DOMINION"
				traits = {
					trait = "trait_cp26_founder_shapeshifter"
					trait = "trait_pc_ocean_preference"
				}
			}
			last_created_species = { save_global_event_target_as = cp26_founder_species }

			create_species = {
				class = "CP26_VORTA"
				portrait = "cp26_vorta_male_01"
				name = "NAME_CP26_Vorta_Species"
				plural = "NAME_CP26_Vorta_Species_Plural"
				adjective = "ADJ_CP26_Vorta_Species"
				name_list = "CP26_DOMINION"
				traits = {
					trait = "trait_cp26_vorta_cloned_administrator"
					trait = "trait_pc_ocean_preference"
				}
			}
			last_created_species = { save_global_event_target_as = cp26_vorta_species }

			create_species = {
				class = "CP26_JEMHADAR"
				portrait = "cp26_jemhadar_male_01"
				name = "NAME_CP26_JemHadar_Species"
				plural = "NAME_CP26_JemHadar_Species_Plural"
				adjective = "ADJ_CP26_JemHadar_Species"
				name_list = "CP26_DOMINION"
				traits = {
					trait = "trait_cp26_jemhadar_engineered_soldier"
					trait = "trait_pc_ocean_preference"
				}
			}
			last_created_species = { save_global_event_target_as = cp26_jemhadar_species }

			create_country = {
				name = "NAME_CP26_The_Dominion"
				species = event_target:cp26_founder_species
				country_id = "dominion"
				type = cp26_dominion
				ship_prefix = "DVS"
				name_list = "CP26_DOMINION"
				authority = auth_cp26_great_link
				civics = { civic = civic_cp26_great_link }
				civics = { civic = civic_cp26_engineered_castes }
				origin = "origin_default"
				flag = {
					icon = { category = "emblem" file = "flag_cp26_dominion.dds" }
					background = { category = "backgrounds" file = "00_solid.dds" }
					colors = { "dark_purple" "black" "null" "null" }
				}
				ethos = {
					ethic = ethic_authoritarian
					ethic = ethic_militarist
					ethic = ethic_xenophobe
				}
				ignore_initial_colony_error = yes
			}

			last_created_country = {
				set_adjective = "ADJ_CP26_Dominion"
				set_graphical_culture = dominion
				set_country_flag = cp26_dominion
				save_global_event_target_as = cp26_dominion_country
				create_leader = {
					class = ruler
					species = event_target:cp26_founder_species
					name = NAME_CP26_Female_Founder
					gender = female
					skill = 7
				}
				set_leader = last_created_leader
			}
		}

		if = {
			limit = { exists = event_target:cp26_dominion_country }
			event_target:cp26_dominion_country = {
				give_technology = { tech = tech_shipyard message = no }
				give_technology = { tech = tech_starbase_starport message = no }
				give_technology = { tech = tech_starbase_starhold message = no }
				give_technology = { tech = tech_planetary_government message = no }
				give_technology = { tech = tech_colonial_centralization message = no }
				give_technology = { tech = tech_corvette_unlock message = no }
				give_technology = { tech = tech_destroyer_unlock message = no }
				give_technology = { tech = tech_cruiser_unlock message = no }
				give_technology = { tech = tech_battleship_unlock message = no }
				give_technology = { tech = tech_phaser_3 message = no }
				give_technology = { tech = tech_torpedo_2 message = no }
				give_technology = { tech = tech_warp_speed_increase_1 message = no }
				give_technology = { tech = tech_warp_speed_increase_2 message = no }
				give_technology = { tech = tech_cp26_dominion_attack_ship message = no }
				give_technology = { tech = tech_cp26_dominion_battle_cruiser message = no }
				give_technology = { tech = tech_cp26_dominion_battleship message = no }
				give_technology = { tech = tech_cp26_dominion_polaron_beam message = no }
				give_technology = { tech = tech_cp26_dominion_polaron_torpedo message = no }
				give_technology = { tech = tech_cp26_jemhadar_cloning message = no }
				give_technology = { tech = tech_cp26_ketracel_white message = no }

				create_ship_design = { design = "NAME_CP26_DominionAttackShip" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_CP26_DominionBattleCruiser" }
				add_ship_design = last_created_design
				create_ship_design = { design = "NAME_CP26_DominionBattleship" }
				add_ship_design = last_created_design

				add_resource = {
					energy = 8000
					minerals = 8000
					alloys = 12000
					food = 6000
					influence = 500
					unity = 3000
				}
			}

			event_target:cp26_dominion_homeworld = {
				create_colony = { owner = event_target:cp26_dominion_country }
				set_controller = event_target:cp26_dominion_country
				set_colony_type = col_default
				set_capital = yes
				clear_blockers = yes
				while = { count = 10 create_pop = { species = event_target:cp26_founder_species } }
				while = { count = 5 create_pop = { species = event_target:cp26_vorta_species } }
				while = { count = 8 create_pop = { species = event_target:cp26_jemhadar_species } }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_city }
				add_district_and_planet_size_if_needed_effect = { district = district_industrial }
				add_district_and_planet_size_if_needed_effect = { district = district_industrial }
				add_district_and_planet_size_if_needed_effect = { district = district_generator }
				add_district_and_planet_size_if_needed_effect = { district = district_mining }
				add_district_and_planet_size_if_needed_effect = { district = district_farming }
				add_building = building_capital
				add_building = building_foundry_1
				add_building = building_research_lab_1
				add_building = building_stronghold
			}

			event_target:cp26_gamma_omarion_system = {
				create_starbase = {
					size = starbase_starport
					owner = event_target:cp26_dominion_country
				}
				starbase = { set_starbase_module = { slot = 1 module = shipyard } }
			}

			event_target:cp26_dominion_country = {
				create_fleet = {
					name = "NAME_CP26_Dominion_First_Fleet"
					effect = {
						while = { count = 6 create_ship = { name = random design = "NAME_CP26_DominionAttackShip" graphical_culture = "dominion_01" } }
						while = { count = 2 create_ship = { name = random design = "NAME_CP26_DominionBattleCruiser" graphical_culture = "dominion_01" } }
						create_ship = { name = random design = "NAME_CP26_DominionBattleship" graphical_culture = "dominion_01" }
						set_owner = prev
						set_location = event_target:cp26_gamma_omarion_system
					}
				}
				create_fleet = {
					name = "NAME_CP26_Dominion_Second_Fleet"
					effect = {
						while = { count = 5 create_ship = { name = random design = "NAME_CP26_DominionAttackShip" graphical_culture = "dominion_01" } }
						while = { count = 2 create_ship = { name = random design = "NAME_CP26_DominionBattleCruiser" graphical_culture = "dominion_01" } }
						create_ship = { name = random design = "NAME_CP26_DominionBattleship" graphical_culture = "dominion_01" }
						set_owner = prev
						set_location = event_target:cp26_gamma_omarion_system
					}
				}
			}
		}
	}
}

##########################################################################
# 2 - Existing-save recovery after the wormhole has already been discovered.
##########################################################################
country_event = {
	id = cp26_dominion.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		has_country_flag = cp26_bajoran_wormhole_discovered
		has_global_flag = cp26_bajoran_wormhole_spawned
		NOT = { has_global_flag = cp26_dominion_spawned }
		exists = event_target:cp26_gamma_omarion_system
	}

	immediate = { country_event = { id = cp26_dominion.1 } }
}
